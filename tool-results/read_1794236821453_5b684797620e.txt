   275→   275→  const setView = useWavely((s) => s.setView);

   279→   279→  const { toast } = useToast();
   280→   280→
   281→   281→  const playlists = useMemo(() => {
   282→   282→    if (mode === "all") return PLAYLISTS;
   283→   283→    if (mode === "local") return PLAYLISTS.filter((p) => p.origin === "local" || p.origin === "mixed");
   284→   284→    return PLAYLISTS.filter((p) => p.origin === "online" || p.origin === "mixed");
   285→   285→  }, [mode]);
   286→   286→
   287→   287→  const jumpBackIds = useMemo(
   288→   288→    () =>
   289→   289→      filterByMode(
   290→   290→        ["t26", "t35", "t17", "t01", "t39", "t31", "t13"]
   291→   291→          .map((id) => trackById.get(id)!)
   292→   292→          .filter(Boolean),
   293→   293→        mode,
   294→   294→      ).map((t) => t.id),
   295→   295→    [mode],
   296→   296→  );
   297→   297→
   298→   298→  const albums = useMemo(() => {

   302→   302→      .map((t) => ALBUMS.find((a) => a.id === t.albumId)!)
   303→   303→      .filter((a, i, arr) => a && arr.findIndex((x) => x.id === a.id) === i)
