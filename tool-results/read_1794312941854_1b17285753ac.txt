    46→function greeting(): string {
    47→  const h = new Date().getHours();
    48→  if (h < 5) return "Good night";
    49→  if (h < 12) return "Good morning";
    50→  if (h < 17) return "Good afternoon";
    51→  if (h < 22) return "Good evening";
    52→  return "Good night";
    53→}

    67→  value,
    68→  onChange,
    69→  ariaLabel,
    70→}: {

    84→            "h-7 rounded-full px-3 text-[11px] font-bold transition",

    88→          )}

   279→            const t = trackById.get(id)!;

   281→              <SongRow
   282→                key={id}

   343→/* ---------------------------------------------------------------- */
   344→/* New for you — online release timeline with a time-window switch   */
   345→/* ---------------------------------------------------------------- */
   346→function NewForYou() {
   347→  const playContext = useWavely((s) => s.playContext);
   348→  const currentTrack = useCurrentTrack();
   349→  const [win, setWin] = useState<7 | 14 | 30>(7);
   350→
   351→  const releases = useMemo(() => {
   352→    const now = Date.now();
   353→    return ALBUMS.filter(
   354→      (a) => a.origin === "online" && a.releasedAt && now - a.releasedAt <= win * 86400000,
   355→    ).sort((a, b) => {
   356→      const fa = FOLLOWED_ARTIST_IDS.includes(a.artistId) ? 0 : 1;
   357→      const fb = FOLLOWED_ARTIST_IDS.includes(b.artistId) ? 0 : 1;
   358→      // followed artists first, then newest release wins
   359→      return fa !== fb ? fa - fb : (b.releasedAt ?? 0) - (a.releasedAt ?? 0);
   360→    });
   361→  }, [win]);
   362→
   363→  return (
   364→    <section>
   365→      <div className="mb-2 flex items-center justify-between gap-3">
   366→        <h3 className="text-sm font-bold text-muted-foreground">New for you</h3>
   367→        <Segmented
   368→          ariaLabel="New release time window"
   369→          value={win}
   370→          onChange={setWin}
   371→          options={[
   372→            { value: 7, label: "1 week" },
   373→            { value: 14, label: "2 weeks" },
   374→            { value: 30, label: "1 month" },
   375→          ]}
   376→        />
   377→      </div>
   378→      {releases.length === 0 ? (
   379→        <p className="rounded-xl bg-wav-surface/60 px-4 py-3 text-xs text-muted-foreground">
   380→          Nothing dropped in this window — try 1 month
   381→        </p>
   382→      ) : (
   383→        <div className="no-scrollbar -mx-4 flex snap-x gap-3 overflow-x-auto px-4 pb-1 md:mx-0 md:px-0">
   384→          {releases.map((a) => (
   385→            <ReleaseCard
   386→              key={a.id}
   387→              cover={a.cover}
   388→              title={a.title}
   389→              meta={`${a.artist} • ${fmtAge(a.releasedAt!)}`}
   390→              playing={currentTrack?.albumId === a.id}
   391→              onPlay={() =>
   392→                playContext(
   393→                  a.title,
   394→                  TRACKS.filter((t) => t.albumId === a.id).map((t) => t.id),
   395→                  0,
   396→                )
   397→              }
   398→            />
   399→          ))}
   400→        </div>
   401→      )}
   402→    </section>
   403→  );
   404→}

   407→/* Daily mixes — top genres rebuilt as AI-style radios               */
   408→/* ---------------------------------------------------------------- */
   409→function DailyMixes() {
   410→  const playContext = useWavely((s) => s.playContext);
   411→
   412→  const mixes = useMemo(() => {
   413→    // top 3 genres by summed playCount across the whole catalogue (local + online)
   414→    const playsByGenre = new Map<string, number>();
   415→    for (const t of TRACKS) playsByGenre.set(t.genre, (playsByGenre.get(t.genre) ?? 0) + t.playCount);
   416→    const topGenres = [...playsByGenre.entries()]
   417→      .sort((a, b) => b[1] - a[1])
   418→      .slice(0, 3)
   419→      .map(([genre]) => genre);
   420→    return topGenres.map((genre, i) => {
   421→      const mix = TRACKS.filter((t) => t.genre === genre)
   422→        .sort((a, b) => b.playCount - a.playCount)
   423→        .slice(0, 8);
   424→      return {
   425→        name: `Daily Mix ${i + 1}`,
   426→        ids: mix.map((t) => t.id),
   427→        // cover = album of the genre's most-played track (mix is sorted by plays)
   428→        cover: albumCover(mix[0]?.albumId ?? ""),
   429→        meta: `${genre} • ${mix.length} songs`,
   430→      };
   431→    });
   432→  }, []);
   433→
   434→  return (
   435→    <section>
   436→      <div className="mb-2 flex items-center justify-between">
   437→        <h3 className="text-sm font-bold text-muted-foreground">Daily mixes</h3>
   438→        <span className="inline-flex items-center gap-1 text-[10px] font-bold uppercase tracking-widest text-primary">
   439→          <Sparkles className="h-3 w-3" /> AI-built
   440→        </span>
   441→      </div>
   442→      <div className="no-scrollbar -mx-4 flex snap-x gap-3 overflow-x-auto px-4 pb-1 md:mx-0 md:px-0">
   443→        {mixes.map((m) => (
   444→          <ReleaseCard
   445→            key={m.name}
   446→            cover={m.cover}
   447→            title={m.name}
   448→            meta={m.meta}
   449→            badge="ai"
   450→            onPlay={() => playContext(m.name, m.ids, Math.floor(Math.random() * m.ids.length))}
   451→          />
   452→        ))}
   453→      </div>
   454→    </section>
   455→  );
   456→}

   510→function MoodsRow() {
   511→  const isOffline = useWavely((s) => s.isOffline);
   512→  const playContext = useWavely((s) => s.playContext);
   513→  const { toast } = useToast();
   514→
   515→  // playable song count per mood (offline shrinks the pool to local/downloaded)
   516→  const counts = useMemo(() => {
   517→    const out: Record<string, number> = {};
   518→    for (const m of MOODS) {
   519→      const candidates = TRACKS.filter((t) => m.genres.includes(t.genre));
   520→      out[m.id] = isOffline
   521→        ? candidates.filter((t) => t.origin === "local" || t.downloaded).length
   522→        : candidates.length;
   523→    }
   524→    return out;
   525→  }, [isOffline]);
   526→
   527→  const playMood = (mood: MoodDef) => {
   528→    let ids = TRACKS.filter((t) => mood.genres.includes(t.genre));
   529→    if (isOffline) ids = ids.filter((t) => t.origin === "local" || t.downloaded);
   530→    if (ids.length === 0) {
   531→      toast({ description: "This mood needs online tracks" });
   532→      return;
   533→    }
   534→    playContext(mood.label, ids.map((t) => t.id), Math.floor(Math.random() * ids.length));
   535→  };
   536→
   537→  return (
   538→    <section>
   539→      <h3 className="mb-2 text-sm font-bold text-muted-foreground">Moods & moments</h3>
   540→      <div className="no-scrollbar -mx-4 flex snap-x gap-2.5 overflow-x-auto px-4 pb-1 md:mx-0 md:px-0">
   541→        {MOODS.map((mood, i) => {
   542→          const Icon = MOOD_ICONS[mood.id];
   543→          return (
   544→            <button
   545→              key={mood.id}
   546→              onClick={() => playMood(mood)}

   572→/* ---------------------------------------------------------------- */
   573→/* Charts — deterministic simulated ranking per week / month         */
   574→/* ---------------------------------------------------------------- */
   575→function ChartsSection() {
   576→  const playContext = useWavely((s) => s.playContext);
   577→  const [period, setPeriod] = useState<"week" | "month">("week");
   578→
   579→  const top5 = useMemo(() => {
   580→    const bucket = period === "week" ? 604800000 : 2592000000;
   581→    const slot = Math.floor(Date.now() / bucket);
   582→    return TRACKS.filter((t) => t.origin === "online")
   583→      .map((t) => {
   584→        // stable pseudo-random bump per period — same ranking all week/month
   585→        const seed = `${t.id}:${period}:${slot}`;
   586→        let h = 0;
   587→        for (const c of seed) h = (h * 31 + c.charCodeAt(0)) >>> 0;
   588→        return { t, score: t.playCount + (h % 37) };
   589→      })
   590→      .sort((a, b) => b.score - a.score)
   591→      .slice(0, 5);
   592→  }, [period]);
   593→
   594→  return (
   595→    <section>
   596→      <div className="mb-2 flex items-center justify-between gap-3">
   597→        <h3 className="text-sm font-bold text-muted-foreground">Top streaming</h3>
   598→        <Segmented
   599→          ariaLabel="Chart period"
   600→          value={period}
   601→          onChange={setPeriod}
   602→          options={[
   603→            { value: "week", label: "This week" },
   604→            { value: "month", label: "This month" },
   605→          ]}
   606→        />
   607→      </div>
   608→      <div className="rounded-2xl bg-wav-surface/50 p-1.5">
   609→        {top5.map(({ t }, i) => (
   610→          <div key={t.id} className="flex items-center gap-3">
   611→            <span
   612→              className={cn(
   613→                "w-6 shrink-0 text-center font-display text-lg font-extrabold tabular-nums",
   614→                i === 0 ? "text-primary" : "text-muted-foreground",
   615→              )}
   616→              aria-hidden="true"
   617→            >
   618→              {i + 1}
   619→            </span>
   620→            <SongRow
   621→              track={t}
   622→              compact
   623→              onPlay={() =>
   624→                playContext(
   625→                  period === "week" ? "Top streaming · this week" : "Top streaming · this month",
   626→                  top5.map((x) => x.t.id),
   627→                  i,
   628→                )
   629→              }
   630→            />
   631→          </div>
   632→        ))}
   633→      </div>
   634→    </section>
   635→  );
   636→}

   638→/* ---------------------------------------------------------------- */
   639→/* Recent artists — from listening history, backfilled by plays      */
   640→/* ---------------------------------------------------------------- */
   641→function RecentArtists() {
   642→  const history = useWavely((s) => s.history);
   643→  const playContext = useWavely((s) => s.playContext);
   644→
   645→  const artists = useMemo(() => {
   646→    const picked: Artist[] = [];
   647→    const seen = new Set<string>();
   648→    // unique artists in most-recent-first history order
   649→    for (const id of history) {
   650→      const t = trackById.get(id);
   651→      if (!t || seen.has(t.artistId)) continue;
   652→      const a = artistById.get(t.artistId);
   653→      if (a) {
   654→        seen.add(a.id);
   655→        picked.push(a);
   656→        if (picked.length >= 8) return picked;
   657→      }
   658→    }
   659→    // thin history → backfill with the most-played artists not picked yet
   660→    if (picked.length < 4) {
   661→      const byPlays = ARTISTS.filter((a) => !seen.has(a.id))
   662→        .map((a) => ({ a, plays: tracksOfArtist(a.id).reduce((sum, t) => sum + t.playCount, 0) }))
   663→        .sort((x, y) => y.plays - x.plays);
   664→      for (const { a } of byPlays) {
   665→        picked.push(a);
   666→        if (picked.length >= 8) break;
   667→      }
   668→    }
   669→    return picked;
   670→  }, [history]);
   671→
   672→  return (
   673→    <section>
   674→      <h3 className="mb-2 text-sm font-bold text-muted-foreground">Recent artists</h3>
   675→      <div className="no-scrollbar -mx-4 flex snap-x gap-3 overflow-x-auto px-4 pb-1 md:mx-0 md:px-0">
   676→        {artists.map((a) => (
   677→          <button
   678→            key={a.id}
   679→            onClick={() => playContext(a.name, tracksOfArtist(a.id).map((t) => t.id), 0)}
   680→            className="flex w-20 shrink-0 snap-start flex-col items-center gap-1.5"
   681→            aria-label={`Play ${a.name}`}
   682→          >
   683→            <span className="relative block h-16 w-16 overflow-hidden rounded-full ring-1 ring-border transition hover:ring-2 hover:ring-primary">
   684→              <Image src={a.image} alt="" fill sizes="64px" className="object-cover" />
   685→            </span>
   686→            <span className="w-full truncate text-center text-[11px] text-muted-foreground">
   687→              {a.name}
   688→            </span>
   689→          </button>
   690→        ))}
   691→      </div>
   692→    </section>
   693→  );
   694→}
