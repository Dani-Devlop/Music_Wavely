   360→class _NextInQueue extends ConsumerWidget {
   361→  const _NextInQueue({required this.queue});
   362→
   363→  final QueueState queue;
   364→
   365→  @override
   366→  Widget build(BuildContext context, WidgetRef ref) {
   367→    final colors = context.wavelyColors;
   368→    final upcoming = <Track>[
   369→      for (var i = queue.currentIndex + 1;
   370→          i < queue.tracks.length && i < queue.currentIndex + 20;
   371→          i++)
   372→        queue.tracks[i],
   373→    ];
   374→    if (upcoming.isEmpty) {
   375→      return Center(
   376→        child: Text(
   377→          'Queue is empty',
   378→          style: WavelyTypography.caption12(color: colors.textTertiary),
   379→        ),
   380→      );
   381→    }
   382→    return ListView.builder(
   383→      itemCount: upcoming.length,
   384→      itemBuilder: (context, index) {
   385→        final track = upcoming[index];
   386→        final absoluteIndex = queue.currentIndex + 1 + index;
   387→        return ListTile(
   388→          dense: true,
   389→          contentPadding: EdgeInsets.zero,
   390→          leading: SizedBox(
   391→            width: 40,
   392→            height: 40,
   393→            child: WavelyCover(
   394→              label: track.title,
   395→              coverPath: track.coverPath,
   396→            ),
   397→          ),
   398→          title: Text(
   399→            track.title,
   400→            maxLines: 1,
   401→            overflow: TextOverflow.ellipsis,
   402→            style:
   403→                WavelyTypography.body14Medium(color: colors.textPrimary),
   404→          ),
   405→          subtitle: Text(
   406→            track.artistOrUnknown,
   407→            maxLines: 1,
   408→            overflow: TextOverflow.ellipsis,
   409→            style: WavelyTypography.caption12(color: colors.textSecondary),
   410→          ),
   411→          trailing: Text(
   412→            track.displayDuration,
   413→            style: WavelyTypography.timerTabular(color: colors.textTertiary),
   414→          ),
   415→          onTap: () => unawaited(
   416→            ref
   417→                .read(queueManagerProvider.notifier)
   418→                .setQueue(List<Track>.of(queue.tracks),
   419→                    startIndex: absoluteIndex),
   420→          ),
   421→        );
   422→      },
   423→    );
   424→  }
   425→}
