    28→function greeting(): string {
    29→  const h = new Date().getHours();
    30→  if (h < 5) return "Good night";
    31→  if (h < 12) return "Good morning";
    32→  if (h < 17) return "Good afternoon";
    33→  if (h < 22) return "Good evening";
    34→  return "Good night";
    35→}

   171→function PlaylistRow({
   172→  name,
   173→  description,
   174→  cover,
   175→  origin,
   176→  trackCount,
   177→  playing,
   178→  onPlay,
   179→  onOpen,
   180→}: {
   181→  name: string;
   182→  description: string;
   183→  cover: string;
   184→  origin: "local" | "online" | "mixed";
   185→  trackCount: number;
   186→  playing?: boolean;
   187→  onPlay: () => void;
   188→  onOpen: () => void;
   189→}) {
   190→  return (
   191→    <div
   192→      role="button"
   193→      tabIndex={0}
   194→      onClick={onOpen}
   195→      onKeyDown={(e) => e.key === "Enter" && onOpen()}
   196→      className="group flex items-center gap-3 rounded-xl p-2 text-left transition hover:bg-accent/60"
   197→    >
   198→      <div className="relative h-[52px] w-[52px] shrink-0 overflow-hidden rounded-xl bg-secondary">
   199→        <Image src={cover} alt="" fill sizes="52px" className="object-cover" />
   200→      </div>
   201→      <div className="min-w-0 flex-1">
   202→        <p className="flex items-center gap-1.5 truncate text-sm font-semibold">
   203→          <span className="truncate">{name}</span>
   204→          {origin === "online" && (
   205→            <span className="inline-flex shrink-0 items-center gap-1 rounded-full bg-sky-400/12 px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wide text-sky-300">
   206→              <Cloud className="h-2.5 w-2.5" /> Online
   207→            </span>
   208→          )}
   209→          {origin === "mixed" && (
   210→            <span className="inline-flex shrink-0 items-center gap-1 rounded-full bg-primary/12 px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wide text-primary">
   211→              <Layers className="h-2.5 w-2.5" /> Mixed
   212→            </span>
   213→          )}
   214→        </p>
   215→        <p className="truncate text-xs text-muted-foreground">
   216→          By Wavely • {trackCount} songs
   217→        </p>
   218→        <p className="truncate text-[11px] text-muted-foreground/70">{description}</p>
   219→      </div>
   220→      <button
   221→        onClick={(e) => {
   222→          e.stopPropagation();
   223→          onPlay();
   224→        }}
   225→        className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full border border-border text-foreground transition hover:border-primary hover:bg-primary hover:text-primary-foreground"
   226→        aria-label={`Play ${name}`}
   227→      >
   228→        {playing ? <EqBars barClassName="bg-primary" /> : <Play className="h-4 w-4 fill-current" />}
   229→      </button>
   230→    </div>
   231→  );
   232→}

   273→export function HomeView() {
   274→  const playContext = useWavely((s) => s.playContext);
   275→  const setView = useWavely((s) => s.setView);
   276→  const currentTrack = useCurrentTrack();
   277→  const mode = useEffectiveMode();
   278→  const isOffline = useWavely((s) => s.isOffline);
   279→  const { toast } = useToast();
   280→
   281→  const playlists = useMemo(() => {
   282→    if (mode === "all") return PLAYLISTS;
   283→    if (mode === "local") return PLAYLISTS.filter((p) => p.origin === "local" || p.origin === "mixed");
   284→    return PLAYLISTS.filter((p) => p.origin === "online" || p.origin === "mixed");
   285→  }, [mode]);
   286→
   287→  const jumpBackIds = useMemo(
   288→    () =>
   289→      filterByMode(
   290→        ["t26", "t35", "t17", "t01", "t39", "t31", "t13"]
   291→          .map((id) => trackById.get(id)!)
   292→          .filter(Boolean),
   293→        mode,
   294→      ).map((t) => t.id),
   295→    [mode],
   296→  );
   297→
   298→  const albums = useMemo(() => {
   299→    const sorted = [...ALBUMS].sort((a, b) => b.year - a.year);
   300→    if (mode === "all") return sorted.slice(0, 4);
   301→    return filterByMode(sorted.flatMap((a) => tracksOfAlbum(a.id)), mode)
   302→      .map((t) => ALBUMS.find((a) => a.id === t.albumId)!)
   303→      .filter((a, i, arr) => a && arr.findIndex((x) => x.id === a.id) === i)
   304→      .slice(0, 4);
   305→  }, [mode]);
   306→
   307→  const localTracks = useMemo(() => TRACKS.filter((t) => t.origin === "local"), []);
   308→  const downloadedTracks = useMemo(() => TRACKS.filter((t) => t.downloaded), []);
   309→  const localStats = useMemo(() => {
   310→    const sec = localTracks.reduce((a, t) => a + t.duration, 0);
   311→    return { count: localTracks.length, time: fmtLongTime(sec) };
   312→  }, [localTracks]);
   313→
   314→  const discoverMix = filterByMode(["t01", "t35", "t13", "t43", "t26", "t39", "t09", "t31", "t22", "t17"], "all");
   315→
   316→  const playlistCover = (id: string) =>
   317→    id === "pl1" ? "/covers/green-circuit.png"
   318→    : id === "pl2" ? "/covers/deep-focus.png"
   319→    : id === "pl3" ? "/covers/persian-echoes.png"
   320→    : id === "pl4" ? "/covers/velvet-static.png"
   321→    : id === "pl5" ? "/covers/neon-horizon.png"
   322→    : id === "pl6" ? "/covers/online-violet-summer.png"
   323→    : id === "pl7" ? "/covers/paper-planes.png"
   324→    : "/covers/golden-hour.png";
   325→
   326→  const eyebrow =
   327→    mode === "local" ? `${greeting()} • Offline player` : mode === "online" ? `${greeting()} • Discover online` : greeting();
   328→
   329→  return (
   330→    <div className="mx-auto w-full max-w-md space-y-7 px-4 pb-10 md:max-w-3xl md:px-8 lg:max-w-5xl">
   331→      {/* ------- sticky header: avatar, actions + source chips ------- */}
   332→      <header className="sticky top-0 z-20 -mx-4 bg-background/85 px-4 pb-3 pt-[max(0.75rem,env(safe-area-inset-top))] backdrop-blur-xl md:-mx-8 md:px-8">
   333→        <div className="flex items-center justify-between gap-3 pt-2">
   334→          <button
   335→            onClick={() => setView("settings")}
   336→            className="relative h-12 w-12 overflow-hidden rounded-full ring-2 ring-primary/40 transition hover:ring-primary"
   337→            aria-label="Open profile"
   338→          >
   339→            <Image src="/wavely/avatar.png" alt="Your profile" fill sizes="48px" className="object-cover" />
   340→          </button>
   341→          <div className="flex items-center gap-2">
   342→            <button
   343→              onClick={() => setView("search")}
   344→              className="flex h-11 w-11 items-center justify-center rounded-full bg-wav-elevated text-foreground transition hover:bg-accent"
   345→              aria-label="Search"
   346→            >
   347→              <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.2" className="h-5 w-5" aria-hidden="true">
   348→                <circle cx="11" cy="11" r="7" />
   349→                <path d="m20 20-3.5-3.5" strokeLinecap="round" />
   350→              </svg>
   351→            </button>
   352→            <button
   353→              onClick={() => {
   354→                useWavely.getState().setLibraryTab("favorites");
   355→                setView("library");
   356→              }}
   357→              className="flex h-11 w-11 items-center justify-center rounded-full bg-wav-elevated text-foreground transition hover:bg-accent"
   358→              aria-label="Favorites"
   359→            >
   360→              <Heart className="h-5 w-5" />
   361→            </button>
   362→          </div>
   363→        </div>
   364→        <SourceModeChips className="mt-3" />
   365→      </header>
   366→
   367→      <OfflineBanner />
   368→
   369→      {/* ------- greeting ------- */}
   370→      <div className="-mt-4">
   371→        <p className="text-[11px] font-semibold uppercase tracking-[0.2em] text-muted-foreground">
   372→          {eyebrow}
   373→        </p>
   374→        <h1 className="mt-1 font-display text-3xl font-extrabold tracking-tight md:text-4xl">
   375→          Hi, Dani
   376→        </h1>
   377→      </div>
   378→
   379→      {/* ------- jump back in (Spotify-style tiles) ------- */}
   380→      <section>
   381→        <SectionHeader title={mode === "online" ? "Pick up where you left off" : "Jump back in"} />
   382→        <div className="grid grid-cols-1 gap-2 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
   383→          {jumpBackIds.slice(0, 6).map((id) => {
   384→            const t = trackById.get(id)!;
   385→            return (
   386→              <QuickTile
   387→                key={id}
   388→                cover={albumCover(t.albumId)}
   389→                title={`${t.title} • ${t.artist}`}
   390→                playing={currentTrack?.id === id}
   391→                onClick={() => playContext("Jump back in", jumpBackIds, jumpBackIds.indexOf(id))}
   392→              />
   393→            );
   394→          })}
   395→        </div>
   396→      </section>
   397→
   398→      {/* ------- LOCAL MODE: offline hero ------- */}
   399→      {mode === "local" && (
   400→        <section>
   401→          <motion.div
   402→            initial={{ opacity: 0, y: 10 }}
   403→            animate={{ opacity: 1, y: 0 }}
   404→            className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-primary/30 via-primary/10 to-transparent p-5"
   405→          >
   406→            <div className="flex items-start gap-4">
   407→              <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-2xl bg-primary text-primary-foreground accent-glow">
   408→                <Smartphone className="h-6 w-6" />
   409→              </div>
   410→              <div className="min-w-0 flex-1">
   411→                <h3 className="font-display text-lg font-bold">Offline player ready</h3>
   412→                <p className="mt-0.5 text-sm text-muted-foreground">
   413→                  {localStats.count} songs on device + {downloadedTracks.length} saved from online
   414→                  • {localStats.time} of music • zero internet needed
   415→                </p>
   416→              </div>
   417→            </div>
   418→            <div className="mt-4 flex flex-wrap items-center gap-2">
   419→              <Button
   420→                className="h-10 gap-2 rounded-full"
   421→                onClick={() => {
   422→                  const ids = filterByMode(TRACKS, "local").map((t) => t.id);
   423→                  playContext("Offline shuffle", ids, Math.floor(Math.random() * ids.length));
   424→                  useWavely.getState().toggleShuffle();
   425→                }}
   426→              >
   427→                <Play className="h-4 w-4 fill-primary-foreground" /> Shuffle offline
   428→              </Button>
   429→              <Button variant="outline" className="h-10 gap-2 rounded-full" onClick={() => seeAll("folders")}>
   430→                <Folder className="h-4 w-4" /> Browse folders
   431→              </Button>
   432→              <Button variant="ghost" className="h-10 rounded-full text-muted-foreground" onClick={() => setView("settings")}>
   433→                Storage settings
   434→              </Button>
   435→            </div>
   436→          </motion.div>
   437→        </section>
   438→      )}
   439→
   440→      {/* ------- curated & trending (all / online) ------- */}
   441→      {mode !== "local" && (
   442→        <section>
   443→          <SectionHeader title={mode === "online" ? "Trending now" : "Curated & trending"} />
   444→          <div className="no-scrollbar -mx-4 flex snap-x snap-mandatory gap-3 overflow-x-auto px-4 pb-1 md:mx-0 md:px-0">
   445→            <CuratedCard
   446→              image="/wavely/curated-1.png"
   447→              badge="ai"
   448→              title="Wavely Weekly"
   449→              subtitle="Fresh picks built from what you play — local and online together."
   450→              onPlay={() => playContext("Wavely Weekly", discoverMix, 0)}
   451→            />
   452→            <CuratedCard
   453→              badge="ai"
   454→              title="Wavely AI DJ"
   455→              subtitle="Ask for a playlist, song facts, or just say “surprise me”."
   456→              onPlay={() => setView("ai")}
   457→            />
   458→            {mode === "online" && (
   459→              <CuratedCard
   460→                image="/covers/online-chrome-city.png"
   461→                badge="online"
   462→                title="Chrome City — full album stream"
   463→                subtitle="Midnight Arcade's new synthpop LP, streaming in 320kbps."
   464→                onPlay={() => playContext("Chrome City", tracksOfAlbum("al9").map((t) => t.id), 0)}
   465→              />
   466→            )}
   467→            <CuratedCard
   468→              image="/wavely/curated-2.png"
   469→              title="Discover Mix"
   470→              subtitle="A scan through genres you haven't touched yet."
   471→              onPlay={() => playContext("Discover Mix", discoverMix.slice().reverse(), 0)}
   472→            />
   473→          </div>
   474→        </section>
   475→      )}
   476→
   477→      {/* ------- top playlists (all modes, filtered) ------- */}
   478→      <section>
   479→        <SectionHeader
   480→          title={mode === "local" ? "Your playlists" : "Top daily playlists"}
   481→          action={
   482→            <Button variant="ghost" size="sm" className="text-xs text-muted-foreground" onClick={() => seeAll("playlists")}>
   483→              See all
   484→            </Button>
   485→          }
   486→        />
   487→        <div className="space-y-0.5">
   488→          {playlists.map((p) => (
   489→            <PlaylistRow
   490→              key={p.id}
   491→              name={p.name}
   492→              description={p.description}
   493→              cover={playlistCover(p.id)}
   494→              origin={p.origin}
   495→              trackCount={p.trackIds.length}
   496→              playing={currentTrack !== undefined && p.trackIds.includes(currentTrack.id)}
   497→              onPlay={() => playContext(p.name, p.trackIds, 0)}
   498→              onOpen={() => seeAll("playlists")}
   499→            />
   500→          ))}
   501→          {playlists.length === 0 && (
   502→            <p className="rounded-xl bg-wav-surface p-4 text-sm text-muted-foreground">
   503→              Nothing here for this filter yet.
   504→            </p>
   505→          )}
