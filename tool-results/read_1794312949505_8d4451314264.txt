    50→    50→  if (h < 17) return "Good afternoon";
    51→    51→  if (h < 22) return "Good evening";
    52→    52→  return "Good night";

   215→   215→function LikedSection({
   216→   216→  favIds,
   217→   217→  onOpen,
   218→   218→}: {
   219→   219→  favIds: string[];
   220→   220→  onOpen: () => void;
   221→   221→}) {
   222→   222→  const playContext = useWavely((s) => s.playContext);
   223→   223→  const toggleShuffle = useWavely((s) => s.toggleShuffle);
   224→   224→
   225→   225→  const shuffleLiked = () => {
   226→   226→    if (favIds.length === 0) return;
   227→   227→    playContext("Liked songs", favIds, Math.floor(Math.random() * favIds.length));
   228→   228→    if (!useWavely.getState().shuffle) toggleShuffle();
   229→   229→  };
   230→   230→
   231→   231→  return (
   232→   232→    <section>
   233→   233→      <SectionHeader
   234→   234→        title="Liked & favorites"
   235→   235→        action={

   237→   237→            See all
   238→   238→          </Button>
   239→   239→        }
   240→   240→      />
   241→   241→      <div className="grid gap-2 sm:grid-cols-2">
   242→   242→        <motion.button
   243→   243→          whileTap={{ scale: 0.985 }}
   244→   244→          onClick={onOpen}
   245→   245→          className="group relative flex min-h-40 flex-col justify-between overflow-hidden rounded-2xl bg-gradient-to-br from-primary/45 via-primary/18 to-transparent p-4 text-left"
   246→   246→          aria-label="Open liked songs"
   247→   247→        >
   248→   248→          <span className="flex h-11 w-11 items-center justify-center rounded-full bg-primary text-primary-foreground shadow-lg accent-glow">
   249→   249→            <Heart className="h-5 w-5 fill-primary-foreground" />
   250→   250→          </span>
   251→   251→          <span className="mt-4 block">
   252→   252→            <span className="block font-display text-lg font-bold">Liked songs</span>
   253→   253→            <span className="mt-0.5 block text-xs text-muted-foreground">
   254→   254→              {favIds.length} songs — whatever you love, in one place
   255→   255→            </span>
   256→   256→          </span>
   257→   257→          <span
   258→   258→            role="button"
   259→   259→            tabIndex={0}
   260→   260→            onClick={(e) => {
   261→   261→              e.stopPropagation();
   262→   262→              shuffleLiked();
   263→   263→            }}
   264→   264→            onKeyDown={(e) => {
   265→   265→              if (e.key === "Enter") {
   266→   266→                e.stopPropagation();
   267→   267→                shuffleLiked();
   268→   268→              }
   269→   269→            }}
   270→   270→            className="absolute bottom-4 right-4 flex h-11 w-11 items-center justify-center rounded-full bg-primary text-primary-foreground opacity-0 shadow-xl transition-all hover:scale-105 focus-visible:opacity-100 group-hover:opacity-100"
   271→   271→            aria-label="Shuffle liked songs"
   272→   272→          >
   273→   273→            <Play className="ml-0.5 h-5 w-5 fill-primary-foreground" />
   274→   274→          </span>
   275→   275→        </motion.button>
   276→   276→
   277→   277→        <div className="flex flex-col justify-center rounded-2xl bg-wav-surface/60 p-1.5">
   278→   278→          {favIds.slice(0, 3).map((id) => {
   279→   279→            const t = trackById.get(id)!;
   280→   280→            return (
   281→   281→              <SongRow
   282→   282→                key={id}
   283→   283→                track={t}
   284→   284→                compact
   285→   285→                onPlay={() => playContext("Liked songs", favIds, favIds.indexOf(id))}
   286→   286→              />
   287→   287→            );
   288→   288→          })}
   289→   289→          {favIds.length > 3 && (
   290→   290→            <button
   291→   291→              onClick={onOpen}
   292→   292→              className="flex items-center gap-1.5 px-2 pb-1.5 pt-1 text-xs font-semibold text-muted-foreground transition hover:text-primary"

   294→   294→              +{favIds.length - 3} more <ChevronRight className="h-3 w-3" />
   295→   295→            </button>
   296→   296→          )}
   297→   297→        </div>
   298→   298→      </div>
   299→   299→    </section>
   300→   300→  );
   301→   301→}

   344→   344→/* New for you — online release timeline with a time-window switch   */
   345→   345→/* ---------------------------------------------------------------- */
   346→   346→function NewForYou() {
   347→   347→  const playContext = useWavely((s) => s.playContext);
   348→   348→  const currentTrack = useCurrentTrack();
   349→   349→  const [win, setWin] = useState<7 | 14 | 30>(7);
   350→   350→
   351→   351→  const releases = useMemo(() => {
   352→   352→    const now = Date.now();
   353→   353→    return ALBUMS.filter(
   354→   354→      (a) => a.origin === "online" && a.releasedAt && now - a.releasedAt <= win * 86400000,
   355→   355→    ).sort((a, b) => {
   356→   356→      const fa = FOLLOWED_ARTIST_IDS.includes(a.artistId) ? 0 : 1;
   357→   357→      const fb = FOLLOWED_ARTIST_IDS.includes(b.artistId) ? 0 : 1;
   358→   358→      // followed artists first, then newest release wins
   359→   359→      return fa !== fb ? fa - fb : (b.releasedAt ?? 0) - (a.releasedAt ?? 0);
   360→   360→    });
   361→   361→  }, [win]);
   362→   362→
   363→   363→  return (
   364→   364→    <section>
   365→   365→      <div className="mb-2 flex items-center justify-between gap-3">
   366→   366→        <h3 className="text-sm font-bold text-muted-foreground">New for you</h3>
   367→   367→        <Segmented
   368→   368→          ariaLabel="New release time window"
   369→   369→          value={win}
   370→   370→          onChange={setWin}
   371→   371→          options={[
   372→   372→            { value: 7, label: "1 week" },
   373→   373→            { value: 14, label: "2 weeks" },
   374→   374→            { value: 30, label: "1 month" },
   375→   375→          ]}
   376→   376→        />
   377→   377→      </div>
   378→   378→      {releases.length === 0 ? (
   379→   379→        <p className="rounded-xl bg-wav-surface/60 px-4 py-3 text-xs text-muted-foreground">
   380→   380→          Nothing dropped in this window — try 1 month
   381→   381→        </p>
   382→   382→      ) : (
   383→   383→        <div className="no-scrollbar -mx-4 flex snap-x gap-3 overflow-x-auto px-4 pb-1 md:mx-0 md:px-0">
   384→   384→          {releases.map((a) => (
   385→   385→            <ReleaseCard
   386→   386→              key={a.id}
   387→   387→              cover={a.cover}
   388→   388→              title={a.title}
   389→   389→              meta={`${a.artist} • ${fmtAge(a.releasedAt!)}`}
   390→   390→              playing={currentTrack?.albumId === a.id}
   391→   391→              onPlay={() =>
   392→   392→                playContext(
   393→   393→                  a.title,
   394→   394→                  TRACKS.filter((t) => t.albumId === a.id).map((t) => t.id),
   395→   395→                  0,
   396→   396→                )
   397→   397→              }
   398→   398→            />
   399→   399→          ))}
   400→   400→        </div>
   401→   401→      )}
   402→   402→    </section>
   403→   403→  );
   404→   404→}

   407→   407→/* Daily mixes — top genres rebuilt as AI-style radios               */
   408→   408→/* ---------------------------------------------------------------- */
   409→   409→function DailyMixes() {
   410→   410→  const playContext = useWavely((s) => s.playContext);
   411→   411→
   412→   412→  const mixes = useMemo(() => {
   413→   413→    // top 3 genres by summed playCount across the whole catalogue (local + online)
   414→   414→    const playsByGenre = new Map<string, number>();
   415→   415→    for (const t of TRACKS) playsByGenre.set(t.genre, (playsByGenre.get(t.genre) ?? 0) + t.playCount);
   416→   416→    const topGenres = [...playsByGenre.entries()]
   417→   417→      .sort((a, b) => b[1] - a[1])
   418→   418→      .slice(0, 3)
   419→   419→      .map(([genre]) => genre);
   420→   420→    return topGenres.map((genre, i) => {
   421→   421→      const mix = TRACKS.filter((t) => t.genre === genre)
   422→   422→        .sort((a, b) => b.playCount - a.playCount)
   423→   423→        .slice(0, 8);
   424→   424→      return {
   425→   425→        name: `Daily Mix ${i + 1}`,
   426→   426→        ids: mix.map((t) => t.id),
   427→   427→        // cover = album of the genre's most-played track (mix is sorted by plays)
   428→   428→        cover: albumCover(mix[0]?.albumId ?? ""),
   429→   429→        meta: `${genre} • ${mix.length} songs`,
   430→   430→      };
   431→   431→    });
   432→   432→  }, []);
   433→   433→
   434→   434→  return (
   435→   435→    <section>
   436→   436→      <div className="mb-2 flex items-center justify-between">
   437→   437→        <h3 className="text-sm font-bold text-muted-foreground">Daily mixes</h3>
   438→   438→        <span className="inline-flex items-center gap-1 text-[10px] font-bold uppercase tracking-widest text-primary">
   439→   439→          <Sparkles className="h-3 w-3" /> AI-built
   440→   440→        </span>
   441→   441→      </div>
   442→   442→      <div className="no-scrollbar -mx-4 flex snap-x gap-3 overflow-x-auto px-4 pb-1 md:mx-0 md:px-0">
   443→   443→        {mixes.map((m) => (
   444→   444→          <ReleaseCard
   445→   445→            key={m.name}
   446→   446→            cover={m.cover}
   447→   447→            title={m.name}
   448→   448→            meta={m.meta}
   449→   449→            badge="ai"
   450→   450→            onPlay={() => playContext(m.name, m.ids, Math.floor(Math.random() * m.ids.length))}
   451→   451→          />
   452→   452→        ))}
   453→   453→      </div>
   454→   454→    </section>
   455→   455→  );
   456→   456→}

   573→   573→/* Charts — deterministic simulated ranking per week / month         */
   574→   574→/* ---------------------------------------------------------------- */
   575→   575→function ChartsSection() {
   576→   576→  const playContext = useWavely((s) => s.playContext);
   577→   577→  const [period, setPeriod] = useState<"week" | "month">("week");
   578→   578→
   579→   579→  const top5 = useMemo(() => {
   580→   580→    const bucket = period === "week" ? 604800000 : 2592000000;
   581→   581→    const slot = Math.floor(Date.now() / bucket);
   582→   582→    return TRACKS.filter((t) => t.origin === "online")
   583→   583→      .map((t) => {
   584→   584→        // stable pseudo-random bump per period — same ranking all week/month
   585→   585→        const seed = `${t.id}:${period}:${slot}`;
   586→   586→        let h = 0;
   587→   587→        for (const c of seed) h = (h * 31 + c.charCodeAt(0)) >>> 0;
   588→   588→        return { t, score: t.playCount + (h % 37) };
   589→   589→      })
   590→   590→      .sort((a, b) => b.score - a.score)
   591→   591→      .slice(0, 5);
   592→   592→  }, [period]);
   593→   593→
   594→   594→  return (
   595→   595→    <section>
   596→   596→      <div className="mb-2 flex items-center justify-between gap-3">
   597→   597→        <h3 className="text-sm font-bold text-muted-foreground">Top streaming</h3>
   598→   598→        <Segmented
   599→   599→          ariaLabel="Chart period"
   600→   600→          value={period}
   601→   601→          onChange={setPeriod}
   602→   602→          options={[
   603→   603→            { value: "week", label: "This week" },
   604→   604→            { value: "month", label: "This month" },
   605→   605→          ]}
   606→   606→        />
   607→   607→      </div>
   608→   608→      <div className="rounded-2xl bg-wav-surface/50 p-1.5">
   609→   609→        {top5.map(({ t }, i) => (
   610→   610→          <div key={t.id} className="flex items-center gap-3">
   611→   611→            <span
   612→   612→              className={cn(
   613→   613→                "w-6 shrink-0 text-center font-display text-lg font-extrabold tabular-nums",
   614→   614→                i === 0 ? "text-primary" : "text-muted-foreground",
   615→   615→              )}
   616→   616→              aria-hidden="true"
   617→   617→            >
   618→   618→              {i + 1}
   619→   619→            </span>
   620→   620→            <SongRow
   621→   621→              track={t}
   622→   622→              compact
   623→   623→              onPlay={() =>
   624→   624→                playContext(
   625→   625→                  period === "week" ? "Top streaming · this week" : "Top streaming · this month",
   626→   626→                  top5.map((x) => x.t.id),
   627→   627→                  i,
   628→   628→                )
   629→   629→              }
   630→   630→            />
   631→   631→          </div>
   632→   632→        ))}
   633→   633→      </div>
   634→   634→    </section>
   635→   635→  );
   636→   636→}

   639→   639→/* Recent artists — from listening history, backfilled by plays      */
   640→   640→/* ---------------------------------------------------------------- */
   641→   641→function RecentArtists() {
   642→   642→  const history = useWavely((s) => s.history);
   643→   643→  const playContext = useWavely((s) => s.playContext);
   644→   644→
   645→   645→  const artists = useMemo(() => {
   646→   646→    const picked: Artist[] = [];
   647→   647→    const seen = new Set<string>();
   648→   648→    // unique artists in most-recent-first history order
   649→   649→    for (const id of history) {
   650→   650→      const t = trackById.get(id);
   651→   651→      if (!t || seen.has(t.artistId)) continue;
   652→   652→      const a = artistById.get(t.artistId);
   653→   653→      if (a) {
   654→   654→        seen.add(a.id);
   655→   655→        picked.push(a);
   656→   656→        if (picked.length >= 8) return picked;
   657→   657→      }
   658→   658→    }
   659→   659→    // thin history → backfill with the most-played artists not picked yet
   660→   660→    if (picked.length < 4) {
   661→   661→      const byPlays = ARTISTS.filter((a) => !seen.has(a.id))
   662→   662→        .map((a) => ({ a, plays: tracksOfArtist(a.id).reduce((sum, t) => sum + t.playCount, 0) }))
   663→   663→        .sort((x, y) => y.plays - x.plays);
   664→   664→      for (const { a } of byPlays) {
   665→   665→        picked.push(a);
   666→   666→        if (picked.length >= 8) break;
   667→   667→      }
   668→   668→    }
   669→   669→    return picked;
   670→   670→  }, [history]);
   671→   671→
   672→   672→  return (
   673→   673→    <section>
   674→   674→      <h3 className="mb-2 text-sm font-bold text-muted-foreground">Recent artists</h3>
   675→   675→      <div className="no-scrollbar -mx-4 flex snap-x gap-3 overflow-x-auto px-4 pb-1 md:mx-0 md:px-0">
   676→   676→        {artists.map((a) => (
   677→   677→          <button
   678→   678→            key={a.id}
   679→   679→            onClick={() => playContext(a.name, tracksOfArtist(a.id).map((t) => t.id), 0)}
   680→   680→            className="flex w-20 shrink-0 snap-start flex-col items-center gap-1.5"
   681→   681→            aria-label={`Play ${a.name}`}
   682→   682→          >
   683→   683→            <span className="relative block h-16 w-16 overflow-hidden rounded-full ring-1 ring-border transition hover:ring-2 hover:ring-primary">
   684→   684→              <Image src={a.image} alt="" fill sizes="64px" className="object-cover" />
   685→   685→            </span>
   686→   686→            <span className="w-full truncate text-center text-[11px] text-muted-foreground">
   687→   687→              {a.name}
   688→   688→            </span>
   689→   689→          </button>
   690→   690→        ))}
   691→   691→      </div>
   692→   692→    </section>
   693→   693→  );
   694→   694→}
